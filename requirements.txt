# --- runtime -----------------------------------------------------------------------------------
# torch is pinned by vLLM: every vLLM release from 0.27.1 through 0.29.0 requires torch==2.13.0
# exactly, so torch and numpy move with vLLM rather than independently (D-009). The default PyPI
# wheel is the CUDA 13 build (2.13.0+cu130, with cudnn-cu13 9.20.0.48 / nccl-cu13 2.29.7), which
# matches the instance driver, so no --index-url is needed.
torch==2.13.0
numpy==2.4.6
datasets==5.0.1
PyYAML==6.0.3

# --- serving -----------------------------------------------------------------------------------
# vLLM 0.27.1 is the last release with built-in bitsandbytes; 0.28.0 moved it to an out-of-tree
# plugin. Serving quantizes in flight to 4-bit NF4 with double quantization (D-008).
# transformers is pinned because vLLM only sets a lower bound on it.
vllm==0.27.1
bitsandbytes==0.50.2
transformers==5.17.0

# --- metrics -----------------------------------------------------------------------------------
# D-018 to D-027. SummaC and AlignScore are ported into metrics/ rather than installed: its package requires
# huggingface-hub<=0.17 and AlignScore's requires torch<2 and transformers<5, so neither can coexist
# with this stack; both are deliberately absent.

# MiniCheck installs from its repository at a pinned commit. The PyPI package named "minicheck" is
